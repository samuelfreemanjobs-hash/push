cmake_minimum_required(VERSION 3.22)
project(DiskLordzThreeLayerDrum VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(DiskLordzThreeLayer INTERFACE)
target_include_directories(DiskLordzThreeLayer INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Source")
add_executable(ThreeLayerRenderer Tools/render_three_layer.cpp)
target_link_libraries(ThreeLayerRenderer PRIVATE DiskLordzThreeLayer)
include(CTest)
if(BUILD_TESTING)
 add_executable(ThreeLayerTests Tests/three_layer_test.cpp)
 target_link_libraries(ThreeLayerTests PRIVATE DiskLordzThreeLayer)
 add_test(NAME ThreeLayerTests COMMAND ThreeLayerTests)
endif()
